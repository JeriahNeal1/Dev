Python>def find_sh_movl_users(lit):
    users = []

    start = max(
        ROM_START,
        lit - 0x404
    )

    for ea in range(start & ~1, lit, 2):
        raw = ida_bytes.get_bytes(ea, 2)

        if not raw:
            continue

        word = int.from_bytes(raw, "big")

        # MOV.L @(disp,PC),Rn
        if (word & 0xF000) != 0xD000:
            continue

        reg = (word >> 8) & 0xF
        disp = word & 0xFF

        target = (
            ((ea + 4) & ~3)
            + (disp << 2)
        )

        if target == lit:
            users.append((ea, reg))

    return users
