Python>def find_sh_movl_users(lit):
    users = []

    start = max(
        ROM_START,
        lit - 0x404
    )

    for ea in range(start & ~1, lit, 2):
        raw = ida_bytes.get_bytes(ea, 2)

        if not raw:
            continue

        word = int.from_bytes(raw, "big")

        # SH:
        # MOV.L @(disp,PC),Rn
        if (word & 0xF000) != 0xD000:
            continue

        reg = (word >> 8) & 0xF
        disp = word & 0xFF

        target = (
            ((ea + 4) & ~3)
            + (disp << 2)
        )

        if target == lit:
            users.append((ea, reg))

    return users

Python>def find_bsr_callers(target):
    callers = []

    for ea in range(
        ROM_START,
        ROM_END - 1,
        2
    ):
        raw = ida_bytes.get_bytes(ea, 2)

        if not raw:
            continue

        word = int.from_bytes(raw, "big")

        # BSR disp12
        if (word & 0xF000) != 0xB000:
            continue

        disp = word & 0x0FFF

        if disp & 0x0800:
            disp -= 0x1000

        dest = ea + 4 + (disp << 1)

        if dest == target:
            callers.append(ea)

    return callers

Python>def find_calls_into_range(lo, hi):
    print(
        f"=== DIRECT BSR TARGETS "
        f"{lo:08X}-{hi:08X} ==="
    )

    found_bsr = set()

    for ea in range(
        ROM_START,
        ROM_END - 1,
        2
    ):
        raw = ida_bytes.get_bytes(ea, 2)

        if not raw:
            continue

        word = int.from_bytes(raw, "big")

        if (word & 0xF000) != 0xB000:
            continue

        disp = word & 0x0FFF

        if disp & 0x0800:
            disp -= 0x1000

        target = ea + 4 + (disp << 1)

        if lo <= target < hi:
            found_bsr.add((ea, target))

    for ea, target in sorted(found_bsr):
        print(
            f"{ea:08X} -> {target:08X}: "
            f"{idc.generate_disasm_line(ea, idc.GENDSM_FORCE_CODE)}"
        )


    print(
        f"\n=== POINTER LITERALS INTO "
        f"{lo:08X}-{hi:08X} ==="
    )

    data = rom_bytes()

    for off in range(0, len(data) - 3, 4):
        value = int.from_bytes(
            data[off:off + 4],
            "big"
        )

        if not (lo <= value < hi):
            continue

        lit = ROM_START + off

        print(
            f"literal {lit:08X} "
            f"-> {value:08X}"
        )

        for user, reg in find_sh_movl_users(lit):
            print(
                f"    MOV.L user "
                f"{user:08X} -> R{reg}"
            )
